# Ear Candy — Podcast Webapp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-hostable podcast webapp with a 3-pane listener UI and a separate admin interface for managing seasons, episodes, and podcast settings.

**Architecture:** Fastify API server with SQLite (port 3001 dev, 3000 prod) paired with a React + Vite SPA (port 5173 dev, served from Fastify in prod). Development uses Docker Compose; production is a single multi-stage Docker image.

**Tech Stack:** TypeScript, Fastify 4, better-sqlite3, React 19, Vite 5, Zustand, Tailwind CSS, Docker, vitest, React Testing Library.

```typescript
export function parseDuration(mmss: string): number {
  const parts = mmss.split(':').map(Number)
  if (parts.length === 3) return parts[0] * 3600 + parts[1] * 60 + parts[2]
  if (parts.length === 2) return parts[0] * 60 + parts[1]
  return 0
}

export function formatDuration(seconds: number): string {
  const h = Math.floor(seconds / 3600)
  const m = Math.floor((seconds % 3600) / 60)
  const s = seconds % 60
  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
  return `${m}:${String(s).padStart(2, '0')}`
}
```

- [ ] **Step 4: Create server/src/db/migrate.ts**

Note: `better-sqlite3`'s `prepare().run()` handles one statement at a time. Each table gets its own call.

```typescript
import type { Database } from 'better-sqlite3'

export function runMigrations(db: Database): void {
  db.prepare(`
    CREATE TABLE IF NOT EXISTS settings (
      podcast_name   TEXT NOT NULL DEFAULT 'My Podcast',
      tagline        TEXT NOT NULL DEFAULT '',
      description    TEXT NOT NULL DEFAULT '',
      cover_art_path TEXT,
      accent_color   TEXT NOT NULL DEFAULT '#5a3ef5'
    )
  `).run()

  db.prepare(`
    CREATE TABLE IF NOT EXISTS seasons (
      id             INTEGER PRIMARY KEY AUTOINCREMENT,
      number         INTEGER NOT NULL,
      title          TEXT NOT NULL,
      description    TEXT NOT NULL DEFAULT '',
      cover_art_path TEXT,
      hidden         INTEGER NOT NULL DEFAULT 0,
      created_at     TEXT NOT NULL DEFAULT (datetime('now'))
    )
  `).run()

  db.prepare(`
    CREATE TABLE IF NOT EXISTS episodes (
      id               INTEGER PRIMARY KEY AUTOINCREMENT,
      season_id        INTEGER NOT NULL REFERENCES seasons(id) ON DELETE CASCADE,
      number           INTEGER NOT NULL,
      title            TEXT NOT NULL,
      description      TEXT NOT NULL DEFAULT '',
      guests           TEXT NOT NULL DEFAULT '',
      tags             TEXT NOT NULL DEFAULT '',
      cover_art_path   TEXT,
      duration_seconds INTEGER NOT NULL DEFAULT 0,
      publish_date     TEXT NOT NULL,
      audio_type       TEXT NOT NULL CHECK(audio_type IN ('upload','url')),
      audio_path       TEXT NOT NULL,
      hidden           INTEGER NOT NULL DEFAULT 0,
      created_at       TEXT NOT NULL DEFAULT (datetime('now')),
      updated_at       TEXT NOT NULL DEFAULT (datetime('now'))
    )
  `).run()

  const { c } = db.prepare('SELECT COUNT(*) as c FROM settings').get() as { c: number }
  if (c === 0) {
    db.prepare('INSERT INTO settings DEFAULT VALUES').run()
  }
}
```

- [ ] **Step 5: Create server/src/db/index.ts**

```typescript
import Database from 'better-sqlite3'
import fs from 'node:fs'
import path from 'node:path'
import { runMigrations } from './migrate.js'

export function initDb(dbPath: string): Database.Database {
  if (dbPath !== ':memory:') {
    const dir = path.dirname(dbPath)
    if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true })
  }
  const db = new Database(dbPath)
  db.pragma('journal_mode = WAL')
  db.pragma('foreign_keys = ON')
  runMigrations(db)
  return db
}
```

```typescript
import Fastify from 'fastify'
import cookie from '@fastify/cookie'
import multipart from '@fastify/multipart'
import staticPlugin from '@fastify/static'
import path from 'node:path'
import type { Database } from 'better-sqlite3'
import { initDb } from './db/index.js'
import { settingsRoute } from './routes/settings.js'

declare module 'fastify' {
  interface FastifyInstance {
    db: Database
  }
}

interface AppOptions {
  dbPath?: string
  logger?: boolean
}

export function buildApp(opts: AppOptions = {}) {
  const app = Fastify({ logger: opts.logger ?? true })
  const dbPath = opts.dbPath ?? path.resolve('data/db.sqlite')
  const db = initDb(dbPath)

  app.decorate('db', db)

  app.register(cookie, {
    secret: process.env.COOKIE_SECRET ?? 'dev-secret-change-in-production'
  })

  app.register(multipart, { limits: { fileSize: 500 * 1024 * 1024 } })

  if (dbPath !== ':memory:') {
    const uploadsDir = path.resolve('data/uploads')
    app.register(staticPlugin, { root: uploadsDir, prefix: '/audio/' })
  }

  app.register(settingsRoute, { prefix: '/api', db })

  return app
}
```

```typescript
import type { FastifyPluginAsync } from 'fastify'
import bcrypt from 'bcrypt'

export const authRoutes: FastifyPluginAsync = async (app) => {
  app.post<{ Body: { password: string } }>('/login', {
    schema: {
      body: {
        type: 'object',
        required: ['password'],
        properties: { password: { type: 'string' } }
      }
    }
  }, async (req, reply) => {
    const hash = process.env.ADMIN_PASSWORD_HASH ?? ''
    const valid = hash.length > 0 && await bcrypt.compare(req.body.password, hash)
    if (!valid) return reply.status(401).send({ error: 'Invalid password' })

    reply.setCookie('admin_session', 'authenticated', {
      signed: true,
      httpOnly: true,
      sameSite: 'strict',
      path: '/',
      maxAge: 60 * 60 * 24 * 7
    })
    return { ok: true }
  })

  app.post('/logout', async (_req, reply) => {
    reply.clearCookie('admin_session', { path: '/' })
    return { ok: true }
  })
}
```

```typescript
import type { FastifyPluginAsync } from 'fastify'
import path from 'node:path'
import fs from 'node:fs'
import { pipeline } from 'node:stream/promises'
import { createWriteStream } from 'node:fs'

export const adminUploadRoute: FastifyPluginAsync = async (app) => {
  async function saveUpload(req: Parameters<typeof app.inject>[0], prefix: string) {
    const data = await (req as any).file()
    if (!data) throw { statusCode: 400, message: 'No file provided' }
    const uploadsDir = path.resolve('data/uploads')
    if (!fs.existsSync(uploadsDir)) fs.mkdirSync(uploadsDir, { recursive: true })
    const ext = path.extname(data.filename as string)
    const filename = `${prefix}-${Date.now()}${ext}`
    await pipeline(data.file, createWriteStream(path.join(uploadsDir, filename)))
    return { filename, url: `/audio/${filename}` }
  }

  app.post('/audio', async (req) => saveUpload(req, 'audio'))
  app.post('/cover', async (req) => saveUpload(req, 'cover'))
}
```

```typescript
export function parseDuration(mmss: string): number {
  const parts = mmss.split(':').map(Number)
  if (parts.length === 3) return parts[0] * 3600 + parts[1] * 60 + parts[2]
  if (parts.length === 2) return parts[0] * 60 + parts[1]
  return 0
}

export function formatDuration(seconds: number): string {
  const h = Math.floor(seconds / 3600)
  const m = Math.floor((seconds % 3600) / 60)
  const s = seconds % 60
  if (h > 0) return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
  return `${m}:${String(s).padStart(2, '0')}`
}
```

```typescript
import { useState, useEffect } from 'react'

type Theme = 'dark' | 'light'

function getInitialTheme(): Theme {
  const stored = localStorage.getItem('theme') as Theme | null
  if (stored) return stored
  return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light'
}

export function useTheme() {
  const [theme, setTheme] = useState<Theme>('dark')

  useEffect(() => {
    const initial = getInitialTheme()
    setTheme(initial)
    document.documentElement.classList.toggle('dark', initial === 'dark')
  }, [])

  const toggle = () => {
    const next: Theme = theme === 'dark' ? 'light' : 'dark'
    setTheme(next)
    localStorage.setItem('theme', next)
    document.documentElement.classList.toggle('dark', next === 'dark')
  }

  return { theme, toggle }
}
```

```typescript
async function get<T>(url: string): Promise<T> {
  const res = await fetch(url)
  if (!res.ok) throw new Error(`API ${res.status}: ${url}`)
  return res.json() as Promise<T>
}

async function mutate<T>(method: string, url: string, body?: unknown): Promise<T> {
  const res = await fetch(url, {
    method,
    headers: body !== undefined ? { 'Content-Type': 'application/json' } : {},
    body: body !== undefined ? JSON.stringify(body) : undefined
  })
  if (!res.ok) throw new Error(`API ${res.status}: ${url}`)
  if (res.status === 204) return undefined as unknown as T
  return res.json() as Promise<T>
}
```

```tsx
import { formatDuration } from '../utils/duration'

interface Props {
  currentTime: number
  duration: number
  onSeek: (t: number) => void
}

export function ProgressBar({ currentTime, duration, onSeek }: Props) {
  const pct = duration > 0 ? (currentTime / duration) * 100 : 0

  function handleClick(e: React.MouseEvent<HTMLDivElement>) {
    const rect = e.currentTarget.getBoundingClientRect()
    onSeek(((e.clientX - rect.left) / rect.width) * duration)
  }

  return (
    <div>
      <div
        className="relative h-1.5 cursor-pointer rounded-full bg-gray-200 dark:bg-gray-700 mb-1"
        onClick={handleClick}
        role="slider"
        aria-valuemin={0}
        aria-valuemax={duration}
        aria-valuenow={Math.floor(currentTime)}
      >
        <div
          className="absolute inset-y-0 left-0 rounded-full"
          style={{ width: `${pct}%`, backgroundColor: 'var(--accent)' }}
        />
        <div
          className="absolute top-1/2 -translate-x-1/2 -translate-y-1/2 h-3.5 w-3.5 rounded-full shadow"
          style={{ left: `${pct}%`, backgroundColor: 'var(--accent)' }}
        />
      </div>
      <div className="flex justify-between text-[11px] tabular-nums text-gray-400 dark:text-gray-500">
        <span>{formatDuration(Math.floor(currentTime))}</span>
        <span>-{formatDuration(Math.floor(duration - currentTime))}</span>
      </div>
    </div>
  )
}
```

```tsx
export function AudioPlayer() {
  const ref = useRef<HTMLAudioElement>(null)
  const {
    currentEpisode, isPlaying, currentTime, duration, playbackRate,
    pause, resume, setCurrentTime, setDuration, setPlaybackRate, skip
  } = usePlayerStore()

  const src = currentEpisode
    ? currentEpisode.audio_type === 'url'
      ? currentEpisode.audio_path
      : `/audio/${currentEpisode.audio_path}`
    : ''

  useEffect(() => {
    if (!ref.current || !currentEpisode) return
    ref.current.src = src
    ref.current.load()
    if (isPlaying) ref.current.play().catch(() => {})
  }, [currentEpisode?.id])

  useEffect(() => {
    if (!ref.current) return
    if (isPlaying) ref.current.play().catch(() => {})
    else ref.current.pause()
  }, [isPlaying])

  useEffect(() => {
    if (ref.current) ref.current.playbackRate = playbackRate
  }, [playbackRate])

  function seek(t: number) {
    if (ref.current) ref.current.currentTime = t
    setCurrentTime(t)
  }

  function handleSkip(s: number) {
    skip(s)
    if (ref.current) ref.current.currentTime = usePlayerStore.getState().currentTime
  }

  function cycleRate() {
    setPlaybackRate(RATES[(RATES.indexOf(playbackRate) + 1) % RATES.length])
  }

  return (
    <div className="px-5 py-3.5">
      <audio
        ref={ref}
        onTimeUpdate={() => setCurrentTime(ref.current?.currentTime ?? 0)}
        onLoadedMetadata={() => setDuration(ref.current?.duration ?? 0)}
      />
      <ProgressBar currentTime={currentTime} duration={duration} onSeek={seek} />
```

```tsx
              <div className="flex justify-end gap-1">
                <IBtn title="Edit" onClick={() => onEditEpisode(ep)}>✎</IBtn>
                <IBtn title={ep.hidden ? 'Show' : 'Hide'} onClick={() => toggleEp(ep)}>◯</IBtn>
                <IBtn title="Delete" danger onClick={() => deleteEp(ep)}>✕</IBtn>
              </div>

          ))}

            </button>
          </div>
        </div>
      )}

function IBtn({ children, onClick, title, danger }: { children: React.ReactNode; onClick: () => void; title: string; danger?: boolean }) {

        ${danger ? 'text-gray-600 hover:bg-red-950/40 hover:text-red-400' : 'text-gray-600 hover:bg-gray-800 hover:text-gray-300'}`}>
      {children}
    </button>
  )
}

function Bdg({ children, v }: { children: React.ReactNode; v: 'green' | 'blue' | 'yellow' }) {
  const cls = { green: 'bg-green-950/50 text-green-400 border-green-800', blue: 'bg-blue-950/50 text-blue-400 border-blue-800', yellow: 'bg-yellow-950/50 text-yellow-400 border-yellow-800' }
  return <span className={`rounded border px-1.5 py-0.5 text-[9px] font-bold ${cls[v]}`}>{children}</span>
}
```

```tsx
import { useEffect, useState } from 'react'
```

```tsx
const inputCls = 'w-full rounded border border-gray-700 bg-gray-800 px-3 py-2 text-xs text-gray-200 focus:border-[color:var(--accent)] focus:outline-none'

function Field({ label, children }: { label: string; children: React.ReactNode }) {
  return (
    <div>
      <label className="mb-1.5 block text-[10px] font-bold uppercase tracking-wide text-gray-500">{label}</label>
      {children}
    </div>
  )

export function SettingsPage() {
  const [form, setForm] = useState<Settings>({ podcast_name: '', tagline: '', description: '', cover_art_path: null, accent_color: '#5a3ef5' })
  const [saving, setSaving] = useState(false)
  const [saved, setSaved] = useState(false)

  useEffect(() => { api.getSettings().then(setForm) }, [])

  function set(k: keyof Settings, v: string) {
    setForm(f => ({ ...f, [k]: v }))
    if (k === 'accent_color') document.documentElement.style.setProperty('--accent', v)
  }

  async function handleCoverUpload(e: React.ChangeEvent<HTMLInputElement>) {
    const file = e.target.files?.[0]
    if (!file) return
    const res = await api.admin.uploadCover(file)
    setForm(f => ({ ...f, cover_art_path: res.url }))
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    setSaving(true)
    try {
      await api.admin.updateSettings(form)
      setSaved(true)
      setTimeout(() => setSaved(false), 2000)
    } finally {
      setSaving(false)
    }
  }

  return (
    <div className="flex flex-1 flex-col overflow-hidden">
      <div className="flex flex-shrink-0 items-center border-b border-gray-800 bg-gray-900 px-4 py-3">
        <h1 className="text-sm font-bold text-white">Podcast Settings</h1>
      </div>
      <form onSubmit={handleSubmit} className="max-w-2xl flex-1 space-y-8 overflow-y-auto p-6">
        <section>
          <h2 className="mb-4 border-b border-gray-800 pb-2 text-xs font-bold uppercase tracking-wide text-gray-500">
            Identity
          </h2>
          <div className="mb-4 grid grid-cols-2 gap-4">
            <Field label="Podcast Name">
              <input value={form.podcast_name} onChange={e => set('podcast_name', e.target.value)} className={inputCls} required />
            </Field>
            <Field label="Tagline">
              <input value={form.tagline} onChange={e => set('tagline', e.target.value)} className={inputCls} />
            </Field>
          </div>
          <Field label="Description">
            <textarea value={form.description} onChange={e => set('description', e.target.value)} rows={3} className={`${inputCls} resize-none`} />
          </Field>
        </section>

        <section>
          <h2 className="mb-4 border-b border-gray-800 pb-2 text-xs font-bold uppercase tracking-wide text-gray-500">
            Appearance
          </h2>
          <div className="grid grid-cols-2 gap-4">
            <Field label="Accent Color">

                  placeholder="#5a3ef5" className={`${inputCls} font-mono`} />
              </div>
            </Field>
            <Field label="Cover Art">
              <div className="flex flex-col items-center rounded-lg border border-dashed border-gray-700 bg-gray-900 p-3 text-center">

                      style={{ background: 'linear-gradient(135deg, var(--accent), color-mix(in srgb, var(--accent) 60%, purple))' }} />
                }

                </label>
              </div>
            </Field>

        </section>

            className="cursor-pointer rounded border border-gray-700 bg-gray-800 px-4 py-2 text-xs font-semibold text-gray-400 hover:text-gray-200">
            Discard

          </button>
        </div>

    </div>
  )
}
```
